# TanStack Query

How to perform asynchronous operations in React.

```tsx title="main.tsx"
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';

const queryClient = new QueryClient();

<QueryClientProvider client={queryClient}>
  <App />

```


## Data Fetching

### Fetch data

Fetch data asynchronously using `useQuery`:

```tsx
import { useQuery, keepPreviousData } from '@tanstack/react-query';
import { fetchTask } from './api';

const {
  data,  // successful query result
  error, // error thrown by queryFn
  isFetching, // true while a fetch is running
} = useQuery({
  queryKey: ['tasks', pagination, filters, sorting],
  queryFn: () => fetchTask(pagination, filters, sorting),
  placeholderData: keepPreviousData,
  staleTime: 1000 * 60 * 5,
});
```

- `queryKey` identifies the query and its cached result. When a value changes, the query runs again.
- `queryFn` is the asynchronous function that fetches data. It must return a promise and throw when the request fails.
- `placeholderData: keepPreviousData` keeps the previous result visible while the next result loads.
- `staleTime: 1000 * 60 * 5` keeps the result fresh for five minutes.

### Show Loading State

**On first load**:
1. Display a loading bar at the top of the page
2. Show a skeleton that matches the main content layout
3. Fetch the data from the server
4. Replace the skeleton with the content when the data loads

**On subsequent loads**:
1. Display a loading bar at the top of the page
2. Keep the previous content visible until the new data loads to avoid layout shifts
3. Dim the content and disable interactions while loading
4. Fetch the data from the server
5. Replace the previous content with the new data when it loads

```tsx
import { cn } from '@/lib/utils';
import { Layout, TopLoadingBar, Table, TableSkeleton, Pagination } from '@/components';

const {
  data,
  error,
  isFetching,
} = useQuery(options);

return (
  <Layout>
    <TopLoadingBar open={isFetching} />

    {isFetching && !data && <TableSkeleton />}

    {(!isFetching || data) && (
      <>
        <Table
          className={cn({
            'pointer-events-none opacity-50': isFetching && data,
          })}
        />
        <Pagination />
      </>
    )}
  </Layout>
);
```

### Error Handling

If fetching fails, throw an error for an error boundary to handle

```tsx
if (error) throw error;
```


## Data Mutation

**Generic flow** (form submission, async actions):
1. Show an inline spinner in the action button while processing
2. Disable the button to prevent multiple submissions
3. Send the request to the server
4. When processing is complete, hide the spinner and re-enable the button
5. Show a success toast on successful completion.

**Error handling**: If submission fails, show an error toast with a user-friendly message.


```tsx title="AddTodo.tsx"
import React, { useState } from 'react';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import axios from 'axios';

const addTodo = async (newTodo: Todo): Promise<Todo> => {
  const response = await axios.post('https://jsonplaceholder.typicode.com/todos', newTodo);
  return response.data;
};

export default function AddTodo() {
  const [title, setTitle] = useState('');
  const queryClient = useQueryClient();

  const mutation = useMutation({
    mutationFn: addTodo,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['todos'] });
    },
  });

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    mutation.mutate({ title, completed: false });
    setTitle('');
  };

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="text"
        value={title}
        onChange={(e) => setTitle(e.target.value)}
        placeholder="Enter todo title"
      />
      <button type="submit">Add Todo</button>
      {mutation.isPending && <p>Adding...</p>}
      {mutation.isError && <p>Error: {(mutation.error as Error).message}</p>}
      {mutation.isSuccess && <p>Todo added ✅</p>}
    </form>
  );
};
```


## Code Snippets

### TopLoadingBar

The component displays a progress bar at the top of the page while data is being fetched. It adapts the bar's progress speed based on past load times.

```tsx title="TopLoadingBar.tsx"
import { useEffect, useState, useCallback, useRef } from 'react';
import ReactTopLoadingBar from 'react-top-loading-bar';

const START_PROGRESS = 1;
const MAX_PROGRESS = 100;
const MAX_LOADING_PROGRESS = 95;
const INTERVAL = 80;
const INITIAL_DURATION = 2000;

type TopLoadingBarProps = {
  open: boolean;
};

export default function TopLoadingBar({ open }: TopLoadingBarProps) {
  const [progress, setProgress] = useState(START_PROGRESS);
  const [duration, setDuration] = useState(INITIAL_DURATION);
  const delta = MAX_LOADING_PROGRESS * (INTERVAL / duration);
  const increaseProgress = useCallback(
    (p: number) => Math.min(p + delta, MAX_LOADING_PROGRESS),
    [delta],
  );
  // needed for tracking past loading bar's completion time
  const isOpened = useRef(open);
  const lastStart = useRef<number | null>(null);
  const pastDurations = useRef<number[]>([]);

  // automatic increase loading bar's progress while open
  useEffect(() => {
    let interval: number;
    if (open) {
      setTimeout(() => setProgress(START_PROGRESS), 0);
      interval = setInterval(() => setProgress(increaseProgress), INTERVAL);
    }
    return () => clearInterval(interval);
  }, [open, increaseProgress]);

  useEffect(() => {
    const durations = pastDurations.current;

    // track loading bar's completion time
    if (open) {
      lastStart.current ??= performance.now();
    } else if (isOpened.current && lastStart.current !== null) {
      durations.push(performance.now() - lastStart.current);
      if (durations.length > 3) {
        durations.shift();
      }
      lastStart.current = null;
    }
    isOpened.current = open;

    // calculate approximate completion time based on past data
    setDuration(
      durations.length > 0
        ? Math.round(durations.reduce((a, b) => a + b, 0) / durations.length)
        : INITIAL_DURATION,
    );
  }, [open]);

  return (
    <ReactTopLoadingBar
      progress={!open ? MAX_PROGRESS : progress}
      shadow={true}
      color="var(--muted-foreground)"
    />
  );
}
```
